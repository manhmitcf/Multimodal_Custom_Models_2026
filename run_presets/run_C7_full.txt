# MARIMO GPU SERVER: TOÀN BỘ QUY TRÌNH CHẠY TỪ ĐẦU (ALL-IN-ONE ONE-SHOT SCRIPT)
# CẤU HÌNH: PRESET 8/8 - C7_full (Full 7 Kênh Động Học [0, 1, 2, 3, 4, 5, 6])
# Mode thời gian: abs_diff (|v1 - v0|)
# Tham số: 2,615,716 (~2.616M params) | FLOPs: ~1.7054 GFLOPs
# Thư mục lưu Checkpoint: checkpoint/Video_C7_full_abs_diff/
# Log file: main_C7_full.log

# ------------------------------------------------------------------------------
# 1. CẤU HÌNH TOKEN VÀ TẢI DATASET (NẾU LÀ MÁY CHỦ MỚI)
# ------------------------------------------------------------------------------
if [ -f "token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' token.txt | head -n 1)
elif [ -f "../token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' ../token.txt | head -n 1)
elif [ -f "/marimo/token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' /marimo/token.txt | head -n 1)
fi

git config --global credential.helper store
if [ -n "$HF_TOKEN" ]; then
    hf auth login --token "$HF_TOKEN" --add-to-git-credential || huggingface-cli login --token "$HF_TOKEN" || true
fi

sudo apt update && sudo apt install git-lfs -y
git lfs install
git config --global lfs.concurrenttransfers 64

if [ ! -d "/marimo/Fish_Feeding_Intensity_Dataset" ]; then
    echo ">>> Đang tải Dataset từ Hugging Face..."
    cd /marimo && git clone https://huggingface.co/datasets/manhmitcf/Fish_Feeding_Intensity_Dataset
fi

# ------------------------------------------------------------------------------
# 2. SẮP XẾP LẠI THƯ MỤC DATASET
# ------------------------------------------------------------------------------
if [ -d "/marimo/Fish_Feeding_Intensity_Dataset/audio/audio" ]; then
    mv /marimo/Fish_Feeding_Intensity_Dataset/audio/audio/* /marimo/Fish_Feeding_Intensity_Dataset/audio/
    rm -rf /marimo/Fish_Feeding_Intensity_Dataset/audio/audio
fi

if [ -d "/marimo/Fish_Feeding_Intensity_Dataset/video/video" ]; then
    mv /marimo/Fish_Feeding_Intensity_Dataset/video/video/* /marimo/Fish_Feeding_Intensity_Dataset/video/
    rm -rf /marimo/Fish_Feeding_Intensity_Dataset/video/video
fi

find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.mp4" | wc -l
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.wav" | wc -l

# ------------------------------------------------------------------------------
# 3. CLONE HOẶC CẬP NHẬT MÃ NGUỒN NHÁNH ABLATION VIDEO
# ------------------------------------------------------------------------------
if [ ! -d "/marimo/Capstone_2026_Fish_Feeding_Intensity" ]; then
    git clone https://github.com/manhmitcf/Multimodal_Custom_Models_2026.git /marimo/Capstone_2026_Fish_Feeding_Intensity
fi
cd /marimo/Capstone_2026_Fish_Feeding_Intensity
git fetch origin
git checkout ablations_final/video_input_channels
git pull origin ablations_final/video_input_channels

cd /marimo/Capstone_2026_Fish_Feeding_Intensity/U_FFIA27K_multimodal
pip install --upgrade pip
pip install -r requirements.txt

# ------------------------------------------------------------------------------
# 4. KIỂM TOÁN TIỀN BAY VÀ CHẠY HUẤN LUYỆN PRESET C7_full
# ------------------------------------------------------------------------------
python main.py --dry-run --preset C7_full --temporal-mode abs_diff

pkill -f "python main.py" || true
sleep 1

nohup python main.py --preset C7_full --temporal-mode abs_diff > main_C7_full.log 2>&1 &
echo ">>> Đã khởi chạy tiến trình huấn luyện C7_full nền!"
echo ">>> Đang mở log theo dõi trực tiếp (nhấn Ctrl+C để thoát xem log, training vẫn tiếp tục chạy)..."
sleep 2
tail -n +1 -f main_C7_full.log
