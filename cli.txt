# ==============================================================================
# 1. CẤU HÌNH TOKEN VÀ TẢI DATASET TRÊN MÁY CHỦ (MARIMO / GPU SERVER)
# ==============================================================================
# Đọc token từ file token.txt (tự động loại bỏ BOM, khoảng trắng thừa)
if [ -f "token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' token.txt | head -n 1)
elif [ -f "../token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' ../token.txt | head -n 1)
elif [ -f "/marimo/token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' /marimo/token.txt | head -n 1)
elif [ -f "/marimo/Capstone_2026_Fish_Feeding_Intensity/token.txt" ]; then
    export HF_TOKEN=$(grep -o 'hf_[a-zA-Z0-9_]\+' /marimo/Capstone_2026_Fish_Feeding_Intensity/token.txt | head -n 1)
fi

if [ -z "$HF_TOKEN" ]; then
    echo "⚠️ CHƯA TÌM THẤY HF_TOKEN! Vui lòng tạo file token.txt (echo 'hf_xxx' > /marimo/token.txt) hoặc chạy: export HF_TOKEN='hf_xxx'"
fi

git config --global credential.helper store
if [ -n "$HF_TOKEN" ]; then
    hf auth login --token "$HF_TOKEN" --add-to-git-credential || huggingface-cli login --token "$HF_TOKEN" || true
fi
sudo apt update && sudo apt install git-lfs -y
git lfs install
git config --global lfs.concurrenttransfers 64
cd /marimo && nohup git clone https://huggingface.co/datasets/manhmitcf/Fish_Feeding_Intensity_Dataset > /marimo/clone.log 2>&1 &
tail -n +1 -f /marimo/clone.log

# ==============================================================================
# 2. SẮP XẾP LẠI THƯ MỤC DATASET
# ==============================================================================
mv /marimo/Fish_Feeding_Intensity_Dataset/audio/audio/* /marimo/Fish_Feeding_Intensity_Dataset/audio/
mv /marimo/Fish_Feeding_Intensity_Dataset/video/video/* /marimo/Fish_Feeding_Intensity_Dataset/video/
rm -rf /marimo/Fish_Feeding_Intensity_Dataset/audio/audio
rm -rf /marimo/Fish_Feeding_Intensity_Dataset/video/video
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.mp4" | wc -l
find /marimo/Fish_Feeding_Intensity_Dataset -type f -iname "*.wav" | wc -l

# ==============================================================================
# 3. HUẤN LUYỆN: Nhánh Thực Nghiệm Audio MLP Dimensions (ablations_final/audio_mlp_dimensions) (400 EPOCHS)
#    (TKEO-STFT-MLP với 10 cấu hình độ sâu & độ rộng: ~0.63M - 4.99M params | embed_dim=384)
#    Artifact Upload Repo: manhmitcf/Results_ablation_audio_mlp_dimensions
# ==============================================================================
if [ ! -d "/marimo/Capstone_2026_Fish_Feeding_Intensity" ]; then
    git clone https://github.com/manhmitcf/Multimodal_Custom_Models_2026.git /marimo/Capstone_2026_Fish_Feeding_Intensity
fi
cd /marimo/Capstone_2026_Fish_Feeding_Intensity
git fetch origin
git checkout ablations_final/audio_mlp_dimensions
git pull origin ablations_final/audio_mlp_dimensions

cd /marimo/Capstone_2026_Fish_Feeding_Intensity/U_FFIA27K_multimodal
pip install --upgrade pip
pip install -r requirements.txt

# --- BƯỚC 3.1: CHẾ ĐỘ TEST & KIỂM TOÁN TIỀN BAY (< 5 giây) ---
python test_audio_ablation.py
python main.py --dry-run

# --- BƯỚC 3.2: CHẠY HUẤN LUYỆN CHÍNH THỨC TRÊN GPU (BACKGROUND) ---
pkill -f "python main.py" || true

# Tùy chọn 1: Chạy preset mặc định cấu hình trong config/train_config.json
nohup python main.py > main.log 2>&1 &
tail -n +1 -f main.log

# Tùy chọn 2: Chạy trực tiếp một preset cụ thể qua cờ --preset (không cần sửa JSON)
# Danh sách 10 presets:
#   1. L0_linear                (0 hidden layers, 2049 -> 384: ~0.79M params)
#   2. W_256_bottleneck         (1 hidden layer,  2049 -> 256 -> 384: ~0.63M params)
#   3. W_512_moderate           (1 hidden layer,  2049 -> 512 -> 384: ~1.25M params)
#   4. W_768_balanced           (1 hidden layer,  2049 -> 768 -> 384: ~1.88M params - Mặc định)
#   5. W_1024_wide              (1 hidden layer,  2049 -> 1024 -> 384: ~2.50M params)
#   6. W_2048_expansion         (1 hidden layer,  2049 -> 2048 -> 384: ~4.99M params)
#   7. L2_pyramidal_1024_512    (2 hidden layers, 2049 -> 1024 -> 512 -> 384: ~2.83M params)
#   8. L2_compact_768_512       (2 hidden layers, 2049 -> 768 -> 512 -> 384: ~2.17M params)
#   9. L2_wide_1024_768         (2 hidden layers, 2049 -> 1024 -> 768 -> 384: ~3.19M params)
#  10. L3_deep_1024_768_512     (3 hidden layers, 2049 -> 1024 -> 768 -> 512 -> 384: ~3.49M params)
#
